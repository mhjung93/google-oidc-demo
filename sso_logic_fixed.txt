  let currentSSOProof = null;
  let currentIdPToken = null; // Step 13 token
  let ssoMetadata = {
    userAddress: null,
    uid: 'user-unique-id-999', // User's ID
    rid: null,               // RP's ID from Token
    r_RP: null,
    arid_i: null,
    auid: null,
    auid_i: null,
    pi_arid_i: null,
    pi_auid: null,
    pi_uid: null,
    salt_fixed: 'wallet-fixed-salt-888' // Fixed salt in wallet
  };

  document.getElementById('mode2ConnectWallet')?.addEventListener('click', async () => {
    mode2Status.innerText = 'Step 1: Connecting Wallet...';
    try {
      if (!window.ethereum) throw new Error('MetaMask not found');
      const accounts = await window.ethereum.request({ method: 'eth_requestAccounts' });
      ssoMetadata.userAddress = accounts[0];
      
      document.getElementById('ssoDataDisplay').style.display = 'block';
      document.getElementById('ssoIntermediateDisplay').innerText = `Step 1. Wallet Connected:\nAddress: ${ssoMetadata.userAddress}`;
      
      document.getElementById('step0PrepMetadata').disabled = false;
      mode2Status.innerText = 'Step 1 Complete. Wallet connected.';
    } catch (err) {
      mode2Status.innerText = `Step 1 Error: ${err.message}`;
    }
  });

  document.getElementById('step0PrepMetadata')?.addEventListener('click', async () => {
    mode2Status.innerText = 'Step 7: RP FE Calculating arid_i...';
    try {
      // 1. Fetch RP Token (rid) from RP BE
      const res = await fetch('/api/mode2/rp_info');
      const rpInfo = await res.json();
      if (!rpInfo.signature) throw new Error('RP not registered. Please register first.');
      ssoMetadata.rid = rpInfo.signature; // Using signature as rid for demo

      // 2. Generate r_RP (Random Scalar)
      ssoMetadata.r_RP = ethers.hexlify(ethers.randomBytes(16));

      // 3. Calculate arid_i = rid * g^r_RP (Simulated)
      ssoMetadata.arid_i = `calc( ${ssoMetadata.rid} * g^${ssoMetadata.r_RP} )`;

      document.getElementById('ssoIntermediateDisplay').innerText += `\n\nStep 7. RP FE Prepared:\nr_RP: ${ssoMetadata.r_RP}\narid_i: ${ssoMetadata.arid_i}`;
      
      document.getElementById('mode2WalletProofs').disabled = false;
      mode2Status.innerText = 'Step 7 Complete. arid_i calculated by RP FE.';
    } catch (err) {
      mode2Status.innerText = `Step 7 Error: ${err.message}`;
    }
  });

  document.getElementById('mode2WalletProofs')?.addEventListener('click', async () => {
    mode2Status.innerText = 'Step 8-9: Wallet Calculating auid and Proofs...';
    try {
      // Wallet uses UID, RID, and its fixed Salt
      // 1. auid = uid * rid * g^Salt (Simulated)
      ssoMetadata.auid = `calc( ${ssoMetadata.uid} * ${ssoMetadata.rid} * g^${ssoMetadata.salt_fixed} )`;
      
      // 2. Mock ZK Proofs pi_auid and pi_uid
      ssoMetadata.pi_auid = `pi_auid-${Math.random().toString(36).substring(2)}`;
      ssoMetadata.pi_uid = `pi_uid-${Math.random().toString(36).substring(2)}`;

      document.getElementById('ssoIntermediateDisplay').innerText += `\n\nStep 8-9. Wallet Data Generated:\nauid: ${ssoMetadata.auid}\npi_auid: ${ssoMetadata.pi_auid}\npi_uid: ${ssoMetadata.pi_uid}`;
      
      document.getElementById('step07RPFECalcAuidi').disabled = false;
      mode2Status.innerText = 'Step 8-9 Complete. Wallet data sent to RP FE.';
    } catch (err) {
      mode2Status.innerText = `Step 8-9 Error: ${err.message}`;
    }
  });

  document.getElementById('step07RPFECalcAuidi')?.addEventListener('click', async () => {
    mode2Status.innerText = 'Step 10: RP FE Calculating auid_i and Proof...';
    try {
      // Formula: auid_i = auid * g^r_RP
      ssoMetadata.auid_i = `calc( ${ssoMetadata.auid} * g^${ssoMetadata.r_RP} )`;
      
      // Dummy proof pi_arid_i
      ssoMetadata.pi_arid_i = `pi_arid_i-${Math.random().toString(36).substring(2)}`;

      document.getElementById('ssoIntermediateDisplay').innerText += `\n\nStep 10. RP FE Final Calculation:\nauid_i: ${ssoMetadata.auid_i}\npi_arid_i: ${ssoMetadata.pi_arid_i}`;
      
      document.getElementById('step1GenerateZKP').disabled = false;
      mode2Status.innerText = 'Step 10 Complete. auid_i and pi_arid_i ready.';
    } catch (err) {
      mode2Status.innerText = `Step 10 Error: ${err.message}`;
    }
  });

  document.getElementById('step1GenerateZKP')?.addEventListener('click', async () => {
    mode2Status.innerText = 'Step 11: Generating Final ZK Proof Package...';
    try {
      const psToken = JSON.parse(localStorage.getItem('mode2_ps_token'));
      // Combine all data for final proof
      currentSSOProof = {
        proof: `zkp-final-${Math.random().toString(36).substring(2)}`,
        publicSignals: { 
          sub: psToken.sub,
          arid_i: ssoMetadata.arid_i,
          auid_i: ssoMetadata.auid_i,
          pi_arid_i: ssoMetadata.pi_arid_i,
          r_RP: ssoMetadata.r_RP,
          pi_auid: ssoMetadata.pi_auid,
          pi_uid: ssoMetadata.pi_uid
        }
      };
      
      document.getElementById('ssoIntermediateDisplay').innerText += `\n\nStep 11. Final ZKP Package Ready:\n${JSON.stringify(currentSSOProof, null, 2)}`;
      
      document.getElementById('step2SubmitToIdP').disabled = false;
      mode2Status.innerText = 'Step 11 Complete. Ready for IdP submission.';
    } catch (err) {
      mode2Status.innerText = `Step 11 Error: ${err.message}`;
    }
  });

  document.getElementById('step2SubmitToIdP')?.addEventListener('click', async () => {
    mode2Status.innerText = 'Step 11-12: Submitting Proof to IdP...';
    try {
      const psToken = JSON.parse(localStorage.getItem('mode2_ps_token'));
      const res = await fetch('http://localhost:4000/sso', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(currentSSOProof)
      });
      const data = await res.json();
      if (data.success) {
        currentIdPToken = data.idpToken;
        
        // Popup for IdP Verification (Step 12 context)
        window.alert(`Step 12: IdP Verification Successful!\n\nIdP Token Issued.\nUser Sub: ${psToken.sub}`);

        document.getElementById('ssoIntermediateDisplay').innerText += `\n\nStep 12. IdP Token Received:\n${JSON.stringify(currentIdPToken, null, 2)}`;
        
        document.getElementById('step25RPFEVerify').disabled = false;
        mode2Status.innerText = 'Step 12 Complete. IdP Token received.';
      } else {
        mode2Status.innerText = `Step 12 Failed: ${data.error}`;
      }
    } catch (err) {
      mode2Status.innerText = `Step 12 Error: ${err.message}`;
    }
  });

  document.getElementById('step25RPFEVerify')?.addEventListener('click', async () => {
    mode2Status.innerText = 'Step 13: RP FE Verifying IdP Token...';
    try {
      if (!currentIdPToken) throw new Error('No IdP Token to verify');
      
      // 1. Verify arid_i match
      if (currentIdPToken.arid_i !== ssoMetadata.arid_i) {
        throw new Error(`arid_i mismatch!\nExpected: ${ssoMetadata.arid_i}\nGot: ${currentIdPToken.arid_i}`);
      }
      
      // 2. Verify auid_i match
      if (currentIdPToken.auid_i !== ssoMetadata.auid_i) {
        throw new Error(`auid_i mismatch!\nExpected: ${ssoMetadata.auid_i}\nGot: ${currentIdPToken.auid_i}`);
      }
      
      // 3. Verify Expiration
      const nowSeconds = Math.floor(Date.now() / 1000);
      if (currentIdPToken.exp <= nowSeconds) {
        throw new Error(`Token expired!\nExp: ${currentIdPToken.exp}\nNow: ${nowSeconds}`);
      }
      
      // 4. Dummy PS Signature Verification
      console.log('[RP FE] Verifying PS Signature of IdP Token...');
      const isSigValid = true; // Simulating PS verify
      if (!isSigValid) throw new Error('Invalid PS Signature on IdP Token');

      document.getElementById('ssoIntermediateDisplay').innerText += `\n\nStep 13. RP FE Verification Success!\n- arid_i OK\n- auid_i OK\n- Expiry OK\n- PS Signature OK`;
      
      document.getElementById('step15NotifyWallet').disabled = false;
      mode2Status.innerText = 'Step 13 Complete. IdP Token is valid.';
    } catch (err) {
      mode2Status.innerText = `Step 13 Error: ${err.message}`;
    }
  });

  document.getElementById('step15NotifyWallet')?.addEventListener('click', async () => {
    mode2Status.innerText = 'Step 15: Notifying Wallet of Success...';
    try {
      // Step 14 context: Internal update after verification
      // Step 15 context: Explicit notification to Wallet
      console.log('[RP FE] Sending notification to Wallet: Auth Success');
      
      // Simulate Snap notification
      document.getElementById('ssoIntermediateDisplay').innerText += `\n\nStep 15. Wallet Notified: Authentication Successful.`;
      
      document.getElementById('step3CompleteRP').disabled = false;
      mode2Status.innerText = 'Step 15 Complete. Wallet state updated.';
    } catch (err) {
      mode2Status.innerText = `Step 15 Error: ${err.message}`;
    }
  });

  document.getElementById('step3CompleteRP')?.addEventListener('click', async () => {
    mode2Status.innerText = 'Step 16-17: Finalizing RP Login...';
    try {
      const res = await fetch('/api/mode2/sso_success', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ idpToken: currentIdPToken })
      });
      const data = await res.json();
      if (data.success) {
        mode2Status.innerText = 'Step 16-17 Complete! RP Login Finalized. You can now visit /me.';
        document.getElementById('ssoIntermediateDisplay').innerText += `\n\nStep 17. RP Session established with full IdP Token!`;
      }
    } catch (err) {
      mode2Status.innerText = `Step 16-17 Error: ${err.message}`;
    }
  });
